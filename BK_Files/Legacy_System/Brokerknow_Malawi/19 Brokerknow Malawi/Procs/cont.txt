	CASE WHEN (@Side = 'S') THEN 14 ELSE @Client_DPA_ END AS SourceEntity_DPA_, 
	CASE WHEN (@Side = 'S') THEN 1 ELSE 8 END AS TargetEntityType_DPA_, 
	CASE WHEN (@Side = 'S') THEN @Client_DPA_ ELSE 14 END AS TargetEntity_DPA_, 
	@LotGross AS TransferAmount, 
	@ContractSettlementDate AS TransferDate, 
	@ChangedBy AS ChangedBy, 
	GETDATE() AS TimeChanged, 
	2 AS InterTransferType_DPA_, 
	@Contract_DPA_ AS Contract_DPA_, 
	0 AS Deleted
	
End

--Update Cash Balance
set @StatementGross = (case when @isCustodian = 1 then 0 else @LotGross end)
set @StatementNet = (case when @Side = 'P' then -(@StatementGross + @TotalCharges) else (@StatementGross - @TotalCharges) end)

UPDATE    ClientBalances
SET              CurrentBal = CurrentBal + @StatementNet
WHERE     (client_DPA_ = @Client_DPA_)
